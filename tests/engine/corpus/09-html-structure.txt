> <style><!-- <style> --></style><p>“x”</p>

# The title is text only: quotes and dashes, never a rule that writes markup.

@ h-title-text-only on=hook-quotes,hook-caps,hook-amp,ordinals
< <title>"Lab" -- it's a test</title><p>"Lab"</p>
> <title>“Lab” — it’s a test</title><p><span class="dquo">“</span>Lab”</p>

@ h-title-text-only-gate on=hook-quotes,hook-caps,hook-amp,ordinals
< <html><head><title>"NASA" &amp; ESA: the 1st launch</title></head><body><p>x</p></body></html>
> <html><head><title>“NASA” &amp; ESA: the 1st launch</title></head><body><p>x</p></body></html>

# An <option> without a value submits its text, so its text is never edited.

@ h-option-without-value
< <select><option>"x" -- it's</option><option selected>a--b</option></select>

@ h-option-with-value
< <select><option value="x">"x" -- it's</option></select>
> <select><option value="x">“x” — it’s</option></select>

@ h-option-with-empty-value
< <select><option value>"x"</option><option value="">'y'</option></select>
> <select><option value>“x”</option><option value="">‘y’</option></select>

@ h-option-implied-end-then-text
< <select><option value=1>"a"<option>"b"</select><p>"c"</p>
> <select><option value=1>“a”<option>"b"</select><p>“c”</p>

# lang="" means the language is unknown: the caller's default, not the parent's.

@ h-lang-empty-is-the-default
< <div lang="de"><p lang="">"x"</p><p>"y"</p></div>
> <div lang="de"><p lang="">“x”</p><p>„y“</p></div>

@ h-lang-empty-is-the-default-fr locale=fr
< <div lang="de"><p lang=" ">"x"</p></div>
> <div lang="de"><p lang=" ">«\u{202F}x\u{202F}»</p></div>

@ h-sorts-lang-empty-still-inherits
< <div lang="de"><p data-sorts-lang="" lang="fr">"x"</p></div>
> <div lang="de"><p data-sorts-lang="" lang="fr">„x“</p></div>

# Comments end at -->, never ->; attributes in single quotes may hold > and ".

@ h-comment-arrow-is-not-the-end
< <!-- a -> "b" --> "c"
> <!-- a -> "b" --> “c”

@ h-single-quoted-attribute-holds-gt
< <a title='x>y "z"'>"q"</a>
> <a title='x>y "z"'>“q”</a>

@ h-cdata-upper-lower
< <![cdata[ "x" ]]> "y"
> <![cdata[ "x" ]]> “y”

# A bogus comment or a doctype inside a paragraph is markup, not a tag: the text
# either side of it is one flow (one pair of quotes, one heading for widows).

@ h-bogus-comment-question-mark-in-a-flow
< <p>"a<?x>b"</p>
> <p>“a<?x>b”</p>

@ h-bogus-comment-bang-in-a-flow
< <h1>one<!x> two three</h1>
> <h1>one<!x> two\u{00A0}three</h1>

@ h-doctype-in-a-flow
< <h1>one<!DOCTYPE x> two three</h1>
> <h1>one<!DOCTYPE x> two\u{00A0}three</h1>

# A CDATA section ends at ]]>, not at ]].

@ h-cdata-ends-at-the-full-marker
< <![CDATA[ a ]] "b" ]]><p>"c"</p>
> <![CDATA[ a ]] "b" ]]><p>“c”</p>

# A raw-text element's end tag is found in any case.

@ h-raw-text-end-tag-upper-case-style
< <STYLE>p{}</STYLE><p>"x"</p>
> <STYLE>p{}</STYLE><p>“x”</p>

@ h-raw-text-end-tag-upper-case-textarea
< <TEXTAREA>"a"</TEXTAREA><p>"x"</p>
> <TEXTAREA>"a"</TEXTAREA><p>“x”</p>

# An element closed by an implied end is closed: a stray end tag of its name later
# closes nothing, and the next end tag still closes its own element.

@ h-stray-end-tag-after-an-implied-end
< <p>a<div data-sorts="off">"b"</p></div><p>"c"</p>
> <p>a<div data-sorts="off">"b"</p></div><p>“c”</p>
